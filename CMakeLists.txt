cmake_minimum_required(VERSION 3.15)

project(Isohypse
    VERSION 0.1.0
    LANGUAGES CXX
)

# ---------------------------------------------------------------------------
# Universal build by default so the plugin loads in both native Apple Silicon
# and Intel builds of Resolume. Override with -DCMAKE_OSX_ARCHITECTURES=arm64
# for a faster single-arch dev build.
#
# This must sit immediately after project() and before any target is defined.
# CMake latches the architecture list when the first target is created, and
# setting it later is silently ignored -- you get an arm64-only binary that the
# build log calls a success. Verify with `lipo`, never with the log.
# ---------------------------------------------------------------------------
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "" FORCE)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Apple deprecated the whole OpenGL API in 10.14 and Resolume's plugin interface
# is still OpenGL, so every SDK call warns. Silencing it is not ignoring a
# problem -- there is no alternative API to move to here.
if(APPLE)
    add_compile_definitions(GL_SILENCE_DEPRECATION)
endif()

if(MSVC)
    add_compile_options(/utf-8)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

option(ISOHYPSE_BUILD_TOOLS "Build the offline render and measurement harness" ON)

# ---------------------------------------------------------------------------
# The Resolume FFGL SDK, as a git submodule under external/.
# ---------------------------------------------------------------------------
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/ffgl/CMakeLists.txt")
    message(FATAL_ERROR
        "FFGL SDK not found. Run: git submodule update --init --recursive")
endif()

set(FFGL_BUILD_EXAMPLE_PLUGINS OFF CACHE BOOL "" FORCE)
add_subdirectory(external/ffgl EXCLUDE_FROM_ALL)

# ---------------------------------------------------------------------------
# The worlds: data/*.png, written by tools/make-worlds.py with the table in
# source/WorldTable.inc. Each plugin carries every one of them -- in the
# bundle's Resources/worlds on macOS, as RCDATA in the DLL on Windows.
# ---------------------------------------------------------------------------
file(GLOB ISOHYPSE_WORLDS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*.png")
if(NOT ISOHYPSE_WORLDS)
    message(FATAL_ERROR "No worlds in data/. Run tools/make-worlds.py.")
endif()

# ---------------------------------------------------------------------------
# Everything both plugins are, as an OBJECT library so both bundles and the
# offline harness are built from exactly the same code.
#
# OBJECT and not STATIC on purpose. CFFGLPluginInfo registers itself from a
# file-scope constructor in SourcePlugin.cpp and EffectPlugin.cpp and nothing
# ever references that object by name, so in an archive the linker is entitled to drop the whole
# translation unit -- and the result is a bundle that loads, exports plugMain,
# and reports that it contains no plugins.
# ---------------------------------------------------------------------------
add_library(isohypse_core OBJECT
    source/Diag.h
    source/Diag.cpp
    source/Controls.h
    source/Controls.cpp
    source/SweepClock.h
    source/SweepClock.cpp
    source/Worlds.h
    source/Worlds.cpp
    source/WorldTable.inc
    source/Target.h
    source/Target.cpp
    source/GLState.h
    source/Shaders.h
    source/Shaders.cpp
    source/Isohypse.h
    source/Isohypse.cpp
)

target_include_directories(isohypse_core PUBLIC source)
target_include_directories(isohypse_core SYSTEM PRIVATE external/stb)
target_link_libraries(isohypse_core PUBLIC ffgl::sdk)
target_compile_definitions(isohypse_core PUBLIC ISOHYPSE_VERSION="${PROJECT_VERSION}")
set_target_properties(isohypse_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

# On Windows/Linux the SDK headers we include pull in <GL/glew.h>. The SDK links
# GLEW privately, so consumers need it too for the include path (and for the
# GLEW_STATIC define the static vcpkg triplet propagates). macOS uses the system
# OpenGL framework and needs none of this.
if(NOT APPLE)
    find_package(GLEW REQUIRED)
    target_link_libraries(isohypse_core PUBLIC GLEW::GLEW)
endif()

# The Windows resource script: one RCDATA per world, named as Worlds.cpp's
# resourceName() names it ("naples-vesuvius.png" is NAPLES_VESUVIUS_PNG).
if(WIN32)
    set(ISOHYPSE_RC "${CMAKE_CURRENT_BINARY_DIR}/worlds.rc")
    set(rc_text "// Generated by CMakeLists.txt from data/*.png -- do not edit.\n")
    foreach(world IN LISTS ISOHYPSE_WORLDS)
        get_filename_component(world_name "${world}" NAME)
        string(TOUPPER "${world_name}" world_id)
        string(REGEX REPLACE "[^A-Z0-9]" "_" world_id "${world_id}")
        file(TO_NATIVE_PATH "${world}" world_native)
        string(REPLACE "\\" "\\\\" world_native "${world_native}")
        string(APPEND rc_text "${world_id} RCDATA \"${world_native}\"\n")
    endforeach()
    file(WRITE "${ISOHYPSE_RC}" "${rc_text}")
endif()

# ---------------------------------------------------------------------------
# The two plugins. FFGL resolves one plugMain per binary, so a source and an
# effect are two bundles -- not one bundle with two entries. SourcePlugin.cpp
# and EffectPlugin.cpp each carry their own CFFGLPluginInfo and are listed in
# their own MODULE target: in the shared library they would register both
# plugins into both bundles.
# ---------------------------------------------------------------------------
function(isohypse_add_plugin target entry bundle_name identifier)
    add_library(${target} MODULE ${entry})
    target_link_libraries(${target} PRIVATE isohypse_core)

    if(APPLE)
        target_sources(${target} PRIVATE ${ISOHYPSE_WORLDS})
        set_source_files_properties(${ISOHYPSE_WORLDS} PROPERTIES
            MACOSX_PACKAGE_LOCATION "Resources/worlds")
        set_target_properties(${target} PROPERTIES
            BUNDLE TRUE
            BUNDLE_EXTENSION "bundle"
            OUTPUT_NAME "${bundle_name}"
            MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in"
            MACOSX_BUNDLE_BUNDLE_NAME "${bundle_name}"
            MACOSX_BUNDLE_GUI_IDENTIFIER "${identifier}"
            MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        )
        # FFGL hosts resolve plugMain by symbol name, so keep default visibility.
        target_link_libraries(${target} PRIVATE
            "-framework OpenGL"
            "-framework Carbon"
            "-framework AppKit"
        )
    else()
        if(WIN32)
            target_sources(${target} PRIVATE "${ISOHYPSE_RC}")
        endif()
        set_target_properties(${target} PROPERTIES OUTPUT_NAME "${bundle_name}")
    endif()
endfunction()

isohypse_add_plugin(IsohypseSource source/SourcePlugin.cpp
    "Isohypse" "com.stoatworks.ffgl.isohypse")
isohypse_add_plugin(IsohypseEffect source/EffectPlugin.cpp
    "Isohypse Over" "com.stoatworks.ffgl.isohypse.over")

# ---------------------------------------------------------------------------
# Offline harness. Whether a ramp crosses the number of contours it should,
# whether a line is as wide on a cliff as on a plain, and whether Olympus
# Mons is where Mars keeps it are measurements, not matters of taste -- so
# this is where the plugin is verified rather than merely previewed.
# ---------------------------------------------------------------------------
if(ISOHYPSE_BUILD_TOOLS AND APPLE)
    add_executable(ihtest tools/ihtest/main.cpp)
    target_include_directories(ihtest PRIVATE tools/ihtest)
    # zlib ships with the OS, which is why the PNG writer is fifty lines here
    # rather than a vendored dependency.
    target_link_libraries(ihtest PRIVATE isohypse_core z
        "-framework OpenGL"
        "-framework AppKit"
    )
    target_compile_definitions(ihtest PRIVATE
        ISOHYPSE_SOURCE_DATA="${CMAKE_CURRENT_SOURCE_DIR}/data")
endif()

# ---------------------------------------------------------------------------
# `cmake --install` drops the bundle straight into the user's Resolume plugin
# folder. Override with -DCMAKE_INSTALL_PREFIX=...
# ---------------------------------------------------------------------------
if(APPLE AND NOT CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    # honour an explicit prefix
elseif(APPLE)
    set(CMAKE_INSTALL_PREFIX "$ENV{HOME}/Documents/Resolume Arena/Extra Effects"
        CACHE PATH "" FORCE)
endif()

install(TARGETS IsohypseSource IsohypseEffect
    LIBRARY DESTINATION "."
    BUNDLE  DESTINATION "."
)
